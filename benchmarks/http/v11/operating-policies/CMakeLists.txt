cmake_minimum_required(VERSION 3.20)
project(doba_operating_policies_benchmark LANGUAGES CXX)
set(DOBA_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../include" CACHE PATH "Headers under test")
option(DOBA_BENCH_SKIP_UNPAUSED_RESUME "Model the optimized Linux resume path" ON)
option(DOBA_BENCH_PUBLISHED_PRESSURE "Model the published Linux pressure flag" ON)
find_package(Threads REQUIRED)
add_executable(operating_activity_benchmark activity.cpp)
target_compile_features(operating_activity_benchmark PRIVATE cxx_std_20)
target_include_directories(operating_activity_benchmark PRIVATE "${DOBA_INCLUDE_DIR}")
target_link_libraries(operating_activity_benchmark PRIVATE Threads::Threads)
add_executable(operating_policies_benchmark main.cpp)
target_compile_features(operating_policies_benchmark PRIVATE cxx_std_20)
target_include_directories(operating_policies_benchmark PRIVATE "${DOBA_INCLUDE_DIR}")
target_link_libraries(operating_policies_benchmark PRIVATE Threads::Threads)
if(DOBA_BENCH_SKIP_UNPAUSED_RESUME)
  target_compile_definitions(operating_policies_benchmark PRIVATE OPTIMIZED_NATIVE)
endif()
if(DOBA_BENCH_PUBLISHED_PRESSURE)
  target_compile_definitions(operating_policies_benchmark PRIVATE PUBLISHED_PRESSURE)
endif()
